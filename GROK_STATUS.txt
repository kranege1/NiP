⚡ QUICKSTART:

1. pip install -r requirements.txt

2. export GROK_API_KEY="dein_api_key"
   (oder: $env:GROK_API_KEY = "dein_api_key" auf Windows PowerShell)

3. python server.py
   (oder: node server.js)

4. curl -X POST http://localhost:3000/api/grok/prompt \
     -H "Content-Type: application/json" \
     -d '{"prompt": "Wer war Einstein?"}'
